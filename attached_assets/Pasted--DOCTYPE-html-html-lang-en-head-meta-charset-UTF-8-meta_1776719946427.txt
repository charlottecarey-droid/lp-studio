<script>
(function() {
  // Dandy locked assumptions
  const DANDY_PRICE = 70;
  const DANDY_RATE = 0.0267;
  const CHAIR_HR = 1;
  const CHAIR_RATE = 375;
  const DANDY_REMAKE_FEE = 0;
  const DANDY_SHIPPING = 0;
  const DANDY_MATERIALS = 0;

  const $ = (id) => document.getElementById(id);

  const inputs = {
    volume: $('volume'),
    price: $('price'),
    remakeRate: $('remakeRate'),
    remakeFee: $('remakeFee'),
    shipping: $('shipping'),
    materials: $('materials')
  };

  const fmt = (n) => {
    const abs = Math.abs(n);
    const sign = n < 0 ? '-' : '';
    if (abs >= 1000) {
      return sign + '$' + abs.toLocaleString('en-US', { maximumFractionDigits: 0 });
    }
    return sign + '$' + abs.toFixed(2);
  };
  const fmtSigned = (n) => {
    const sign = n > 0 ? '+' : (n < 0 ? '-' : '');
    return sign + '$' + Math.abs(n).toLocaleString('en-US', { maximumFractionDigits: 2, minimumFractionDigits: 2 });
  };
  const fmtK = (n) => {
    const abs = Math.abs(n);
    const sign = n < 0 ? '-' : '';
    return sign + '$' + abs.toLocaleString('en-US', { maximumFractionDigits: 0 });
  };
  const fmtDelta = (n) => {
    const sign = n > 0 ? '+' : (n < 0 ? '-' : '');
    return sign + '$' + Math.abs(n).toLocaleString('en-US', { maximumFractionDigits: 0 });
  };

  function calc() {
    const volume = Math.max(0, +inputs.volume.value || 0);
    const price = Math.max(0, +inputs.price.value || 0);
    const rate = Math.max(0, (+inputs.remakeRate.value || 0) / 100);
    const remakeFee = Math.max(0, +inputs.remakeFee.value || 0);
    const shipping = Math.max(0, +inputs.shipping.value || 0);
    const materials = Math.max(0, +inputs.materials.value || 0);

    // Budget side
    const budgetBlendedRemake = CHAIR_HR * CHAIR_RATE + remakeFee + shipping + materials;
    const budgetLoad = rate * budgetBlendedRemake;
    const budgetTrue = price + budgetLoad;

    // Dandy side
    const dandyBlendedRemake = CHAIR_HR * CHAIR_RATE + DANDY_REMAKE_FEE + DANDY_SHIPPING + DANDY_MATERIALS;
    const dandyLoad = DANDY_RATE * dandyBlendedRemake;
    const dandyTrue = DANDY_PRICE + dandyLoad;

    // Comparison
    const perUnitDelta = dandyTrue - budgetTrue;
    const stickerGap = DANDY_PRICE - price;
    const trueGap = dandyTrue - budgetTrue;
    const annualBudget = budgetTrue * volume;
    const annualDandy = dandyTrue * volume;
    const annualDelta = annualDandy - annualBudget;

    // Update comparison table
    $('compBudgetPrice').textContent = fmt(price) + ' sticker';
    $('rowStickerBudget').textContent = fmt(price);
    $('rowStickerDandy').textContent = fmt(DANDY_PRICE);
    $('rowRateBudget').textContent = (rate * 100).toFixed(1) + '%';
    $('rowRemakeCostBudget').textContent = fmt(budgetBlendedRemake);
    $('rowLoadBudget').textContent = fmtSigned(budgetLoad);
    $('rowLoadDandy').textContent = fmtSigned(dandyLoad);
    $('rowTrueBudget').textContent = fmt(budgetTrue);
    $('rowTrueDandy').textContent = fmt(dandyTrue);

    // Annual block
    $('annualBudget').textContent = fmtK(annualBudget);
    $('annualDandy').textContent = fmtK(annualDandy);

    const deltaEl = $('annualDelta');
    const deltaLabel = $('annualDeltaLabel');
    if (annualDelta < -500) {
      deltaLabel.textContent = 'Annual SAVINGS switching to Dandy';
      deltaEl.textContent = fmtDelta(-annualDelta);
      deltaEl.className = 'value win';
    } else if (annualDelta > 500) {
      deltaLabel.textContent = 'Annual premium to switch to Dandy';
      deltaEl.textContent = fmtDelta(annualDelta);
      deltaEl.className = 'value loss';
    } else {
      deltaLabel.textContent = 'Annual cost impact — effectively a wash';
      deltaEl.textContent = fmtDelta(annualDelta);
      deltaEl.className = 'value';
    }

    // Verdict block — the hero output
    const verdictBig = $('verdictBig');
    const verdictEyebrow = $('verdictEyebrow');
    const verdictStatement = $('verdictStatement');

    const absGap = Math.abs(trueGap);
    const reductionPct = stickerGap !== 0 ? Math.round((1 - absGap / Math.abs(stickerGap)) * 100) : 0;

    if (trueGap < -0.5) {
      // Dandy actually cheaper on true cost
      verdictEyebrow.textContent = 'Dandy is cheaper per unit';
      verdictBig.textContent = fmtSigned(trueGap) + '/crown';
      verdictBig.className = 'verdict-big win';
      verdictStatement.innerHTML = `A <strong>${fmt(Math.abs(stickerGap))}</strong> sticker gap in your lab's favor becomes a <strong>${fmt(absGap)} per unit advantage for Dandy</strong> once you count remake cost, chair time, and shipping honestly. At ${volume.toLocaleString()} units/year, that's <strong>${fmtK(Math.abs(annualDelta))} in annual savings</strong>.`;
    } else if (trueGap < 0.5) {
      // Effectively a wash
      verdictEyebrow.textContent = 'Effectively a wash';
      verdictBig.textContent = fmtSigned(trueGap) + '/crown';
      verdictBig.className = 'verdict-big';
      verdictBig.style.color = '#0a0a0a';
      verdictStatement.innerHTML = `What looks like a <strong>${fmt(Math.abs(stickerGap))}</strong> sticker gap lands at <strong>${fmt(absGap)} per unit</strong> on true cost. That's ${Math.max(0, reductionPct)}% of the gap gone. Add one rush fee, one complex case, or the admin time this calculator doesn't count — and Dandy comes out ahead.`;
    } else {
